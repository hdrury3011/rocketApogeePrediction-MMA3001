main

Entry point: runs the whole apogee-prediction study end to end.

Usage (from the repository root, with the virtual environment active)::

python src/main.py --data data/project_data_set.csv

Steps

  1. Verification: integrators against the analytical constant-k solution.
  2. Convergence and cost of Euler and RK4 on the full model (ISA + C_D table).
  3. Calibration of the unknown rocket mass to the 3450 m no-airbrake apogee.
  4. Validation against the 11000 ft airbrake apogee and flap/deployment study.
  5. Sensitivity, modelling-assumption and performance studies.

Figures are saved to /figures and a results summary to /results.md.

 43def style_axes(ax, title=None, xlabel=None, ylabel=None):
 44    """Apply a quiet, consistent look: recessive grid, no top/right spines."""
 45    ax.set_facecolor(SURFACE)
 46    ax.grid(True, color=GRID, linewidth=0.8)
 47    ax.set_axisbelow(True)
 48    for side in ('top', 'right'):
 49        ax.spines[side].set_visible(False)
 50    for side in ('left', 'bottom'):
 51        ax.spines[side].set_color(INK2)
 52    ax.tick_params(colors=INK2, labelsize=9)
 53    # Log axes: keep decade labels only (minor tick labels overlap).
 54    ax.xaxis.set_minor_formatter(NullFormatter())
 55    ax.yaxis.set_minor_formatter(NullFormatter())
 56    if title:
 57        ax.set_title(title, color=INK, fontsize=11, loc='left')
 58    if xlabel:
 59        ax.set_xlabel(xlabel, color=INK2, fontsize=10)
 60    if ylabel:
 61        ax.set_ylabel(ylabel, color=INK2, fontsize=10)

 70def timed(fn, repeats=3):
 71    """Run fn() several times; return (result, best wall time in seconds)."""
 72    best, out = np.inf, None
 73    for _ in range(repeats):
 74        t0 = time.perf_counter()
 75        out = fn()
 76        best = min(best, time.perf_counter() - t0)
 77    return out, best

 87def observed_orders(dts, errs):
 88    return [float(np.log(errs[i] / errs[i + 1]) / np.log(dts[i] / dts[i + 1]))
 89            for i in range(len(errs) - 1)]

 95def run_verification(plt, figdir):
 96    """Euler/RK4 error against the closed-form apogee (constant rho and C_D)."""
 97    rho, cd0, k = 1.225, 0.40, 1.0e-4
 98    mass = rho * cd0 * S_REF / (2 * k)                    # gives k exactly
 99    atm = constant_atmosphere(rho, 340.29)
100    exact = analytic_apogee_height(V_BURNOUT, k, G)
101    dts = [0.4 * 0.5**n for n in range(8)]
102
103    out = {}
104    for name, step in (('Euler', euler_step), ('RK4', rk4_step)):
105        errs, evals = [], []
106        for dt in dts:
107            counter = [0]
108            _, h = simulate_apogee(mass, lambda m: cd0, atm, V_BURNOUT,
109                                   H_BURNOUT, dt=dt, step=step, counter=counter)
110            errs.append(abs((h - H_BURNOUT) - exact))
111            evals.append(counter[0])
112        out[name] = dict(errs=errs, evals=evals)
113
114    if plt is not None:
115        fig, (ax1, ax2) = new_figure(plt, 2, 10.5, 4.4)
116        for name, color in (('Euler', ORANGE), ('RK4', BLUE)):
117            ax1.loglog(dts, out[name]['errs'], '-o', color=color, lw=2, ms=8, label=name)
118            ax2.loglog(out[name]['evals'], out[name]['errs'], '-o', color=color,
119                       lw=2, ms=8, label=name)
120        dref = np.array([dts[0], dts[-1]])
121        ax1.loglog(dref, out['Euler']['errs'][0] * (dref / dts[0]), '--', color=INK2,
122                   lw=1, label='slope 1 and 4 (reference)')
123        ax1.loglog(dref, out['RK4']['errs'][2] * (dref / dts[2])**4, '--', color=INK2, lw=1)
124        style_axes(ax1, 'Apogee error vs step size (analytical benchmark)',
125                   'step size dt (s)', 'absolute apogee error (m)')
126        style_axes(ax2, 'Apogee error vs cost', 'right-hand-side evaluations', None)
127        for ax in (ax1, ax2):
128            ax.legend(frameon=False, fontsize=9, labelcolor=INK2)
129        fig.tight_layout()
130        fig.savefig(figdir / 'fig1_verification.png', dpi=160, facecolor=SURFACE)
131        plt.close(fig)
132
133    out['dts'] = dts
134    out['exact'] = exact
135    # RK4 is only fitted over the first four steps: below dt ~ 0.05 s the
136    # error reaches the round-off floor and the slope flattens.
137    out['orders'] = {'Euler': observed_orders(dts[:6], out['Euler']['errs'][:6]),
138                     'RK4': observed_orders(dts[:4], out['RK4']['errs'][:4])}
139    return out

145def run_calibration(model_lin):
146    """Calibrate mass with bisection and Brent; compare cost."""
147    res = {}
148    for method in ('bisect', 'brent'):
149        (mass, n), t = timed(lambda m=method: calibrate_mass(
150            APOGEE_NO_BRAKES, model_lin, isa, V_BURNOUT, H_BURNOUT,
151            method=m, xtol=1e-4), repeats=1)
152        res[method] = dict(mass=mass, n_sim=n, time=t)
153    return res

156def run_full_convergence(model_lin, mass):
157    """RK4 vs Euler against a tight DOP853 reference on the full model."""
158    _, h_ref = reference_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT)
159    dts = [0.8 * 0.5**n for n in range(7)]
160    out = {'h_ref': h_ref, 'dts': dts}
161    for name, step in (('Euler', euler_step), ('RK4', rk4_step)):
162        errs, evals, times = [], [], []
163        for dt in dts:
164            counter = [0]
165            (_, h), t = timed(lambda: simulate_apogee(
166                mass, model_lin, isa, V_BURNOUT, H_BURNOUT, dt=dt, step=step,
167                counter=counter), repeats=1)
168            errs.append(abs(h - h_ref))
169            evals.append(counter[0])
170            times.append(t)
171        out[name] = dict(errs=errs, evals=evals, times=times)
172    out['orders'] = {n: observed_orders(dts[:5], out[n]['errs'][:5]) for n in ('Euler', 'RK4')}
173    return out

176def run_smoothness_study(model_lin, model_cub, mass):
177    """RK4 error vs dt for three C_D models of different smoothness.
178
179    A piecewise-linear table has slope jumps at every Mach node and a cubic
180    table is clamped at Mach 0.1; RK4's fourth-order accuracy assumes a
181    smooth right-hand side. A smooth analytic C_D is the control case.
182    """
183    smooth = lambda m: 0.4 + 0.05 * np.sin(3.0 * m)          # noqa: E731
184    models = {'linear table': model_lin, 'cubic table': model_cub,
185              'smooth C_D (control)': smooth}
186    dts = [0.8 * 0.5**n for n in range(7)]
187    out = {'dts': dts}
188    for name, model in models.items():
189        h_ref = reference_apogee(mass, model, isa, V_BURNOUT, H_BURNOUT,
190                                 rtol=1e-13, atol=1e-11)[1]
191        errs = [abs(simulate_apogee(mass, model, isa, V_BURNOUT, H_BURNOUT,
192                                    dt=dt, step=rk4_step)[1] - h_ref)
193                for dt in dts]
194        out[name] = errs
195    return out

198def run_validation(model_lin, mass):
199    """Apogee vs flap, equivalent flap and deployment delay for the airbrake target."""
200    flaps = [0, 25, 50, 75, 100]
201    sweep = [simulate_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT, flap=f)[1]
202             for f in flaps]
203    flap_eq, _ = flap_for_apogee(APOGEE_WITH_BRAKES, mass, model_lin, isa,
204                                 V_BURNOUT, H_BURNOUT)
205
206    def apogee_with_delay(td):
207        return reference_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT,
208                                flap=lambda t: 100.0 if t >= td else 0.0,
209                                rtol=1e-9, atol=1e-8)[1]
210
211    t_nb, _ = reference_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT)
212    td, _ = bisect(lambda td: apogee_with_delay(td) - APOGEE_WITH_BRAKES,
213                   0.0, t_nb, xtol=0.01)
214    return dict(flaps=flaps, sweep=sweep, flap_eq=flap_eq, delay=td,
215                t_apogee_no_brakes=t_nb)

218def run_sensitivity(model_lin, mass, base):
219    """One-at-a-time sensitivity of the (no-airbrake) apogee."""
220    def scaled_atm(s):
221        return lambda h: (s * isa(h)[0], isa(h)[1])
222
223    def apogee(**kw):
224        p = dict(mass=mass, v0=V_BURNOUT, cd=model_lin, atm=isa, area=S_REF)
225        p.update(kw)
226        return simulate_apogee(p['mass'], p['cd'], p['atm'], p['v0'], H_BURNOUT,
227                               area=p['area'])[1]
228
229    def cd_scaled(s):
230        return DragModel(model_lin.mach, model_lin.flap, model_lin.cd * s)
231
232    cases = [
233        ('mass +/-10 %', dict(mass=mass * 1.10), dict(mass=mass * 0.90)),
234        ('burnout speed +/-5 m/s', dict(v0=V_BURNOUT + 5), dict(v0=V_BURNOUT - 5)),
235        ('C_D +/-5 %', dict(cd=cd_scaled(1.05)), dict(cd=cd_scaled(0.95))),
236        ('reference area +/-2 %', dict(area=S_REF * 1.02), dict(area=S_REF * 0.98)),
237        ('air density +/-5 %', dict(atm=scaled_atm(1.05)), dict(atm=scaled_atm(0.95))),
238    ]
239    rows = [(label, apogee(**up) - base, apogee(**dn) - base) for label, up, dn in cases]
240    return rows

243def run_assumptions(model_lin, model_cub, mass, base):
244    """How much do modelling choices move the apogee?"""
245    out = {}
246    out['cubic_vs_linear'] = [
247        (f, simulate_apogee(mass, model_cub, isa, V_BURNOUT, H_BURNOUT, flap=f)[1]
248         - simulate_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT, flap=f)[1])
249        for f in (0.0, 50.0, 100.0)]
250
251    # Constant-property (analytical) approximations of the same flight.
252    a0 = isa(H_BURNOUT)[1]
253    cd_burnout = model_lin(V_BURNOUT / a0, 0.0)
254    cd_mid = model_lin(0.5, 0.0)
255    rho_burnout = isa(H_BURNOUT)[0]
256    rho_mid = isa(0.5 * (H_BURNOUT + base))[0]
257    variants = []
258    for label, rho, cd in (('rho at burnout, C_D at burnout Mach', rho_burnout, cd_burnout),
259                           ('rho at mid-coast, C_D at Mach 0.5', rho_mid, cd_mid),
260                           ('sea-level rho, C_D at Mach 0.5', 1.225, cd_mid)):
261        k = rho * cd * S_REF / (2 * mass)
262        variants.append((label, H_BURNOUT + analytic_apogee_height(V_BURNOUT, k, G)))
263    out['constant'] = variants
264    out['vacuum'] = H_BURNOUT + V_BURNOUT**2 / (2 * G)
265
266    # Interpretation of the 3450 m target: altitude vs height gained.
267    m_gain, _ = calibrate_mass(H_BURNOUT + APOGEE_NO_BRAKES, model_lin, isa,
268                               V_BURNOUT, H_BURNOUT, bracket=(5.0, 2000.0))
269    out['mass_if_height_gain'] = m_gain
270    for dtarget in (-50.0, 50.0):
271        m, _ = calibrate_mass(APOGEE_NO_BRAKES + dtarget, model_lin, isa,
272                              V_BURNOUT, H_BURNOUT)
273        out[f'mass_target_{dtarget:+.0f}'] = m
274    return out

277def run_performance(model_lin, mass):
278    """Baseline vs optimised right-hand side, fixed-step vs adaptive."""
279    # Baseline: flap passed as a function of time, which forces the full
280    # C_D(Mach, flap) lookup (all table columns) on every evaluation.
281    counter_b, counter_o = [0], [0]
282    (_, t_b) = timed(lambda: simulate_apogee(
283        mass, model_lin, isa, V_BURNOUT, H_BURNOUT, flap=lambda t: 0.0,
284        counter=counter_b))
285    (_, t_o) = timed(lambda: simulate_apogee(
286        mass, model_lin, isa, V_BURNOUT, H_BURNOUT, flap=0.0, counter=counter_o))
287    n_evals = counter_o[0] // 3                      # evaluations per run
288
289    _, t_ref = timed(lambda: reference_apogee(
290        mass, model_lin, isa, V_BURNOUT, H_BURNOUT, rtol=1e-8, atol=1e-6))
291
292    prof = cProfile.Profile()
293    prof.enable()
294    simulate_apogee(mass, model_lin, isa, V_BURNOUT, H_BURNOUT,
295                    flap=lambda t: 0.0)
296    prof.disable()
297    buf = io.StringIO()
298    pstats.Stats(prof, stream=buf).sort_stats('cumulative').print_stats(8)
299    return dict(t_baseline=t_b, t_optimised=t_o, n_evals=n_evals,
300                t_adaptive=t_ref, profile=buf.getvalue())

398def main(argv=None):

407    if not data_path.is_file():
408        print(f"error: data file not found: {data_path}\n"
409              "Place the CFD csv there or pass --data PATH.", file=sys.stderr)
410        return 1
411
412    plt = None
413    figdir = Path(args.outdir) / 'figures'
414    if not args.no_plots:
415        import matplotlib
416        matplotlib.use('Agg')
417        import matplotlib.pyplot as plt
418        figdir.mkdir(parents=True, exist_ok=True)
419    Path(args.outdir).mkdir(parents=True, exist_ok=True)
420
421    mach, flap, cd = load_cd_table(data_path)
422    model_lin = DragModel(mach, flap, cd, 'linear')
423    model_cub = DragModel(mach, flap, cd, 'cubic')
